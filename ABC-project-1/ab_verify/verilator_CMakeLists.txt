cmake_minimum_required(VERSION 3.20)
project(ab_verification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB generated CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/V*.cpp")
list(FILTER generated EXCLUDE REGEX "__ALL\\.cpp$")
add_executable(ab_sim ${generated} ${VROOT}/include/verilated.cpp ${VROOT}/include/verilated_timing.cpp ${VROOT}/include/verilated_threads.cpp)
target_include_directories(ab_sim PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VROOT}/include ${VROOT}/include/vltstd)
target_compile_definitions(ab_sim PRIVATE VL_TIME_CONTEXT VM_SC=0 VM_TRACE=0 VM_COVERAGE=0)
target_compile_options(ab_sim PRIVATE /O2 /EHsc /bigobj)
